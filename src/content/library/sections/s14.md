Веб-интерфейс из ТЗ-28 ложится на React целиком: всё состояние живёт в одном компоненте `App`, а семь компонентов получают данные только через пропсы. Ниже рабочий набросок: код прогнан в эмуляции браузера, но в вашем репозитории его пока нет.

### Что берём из проекта

- Приложение хранит продукты и их сроки годности и умеет спрашивать рецепт у AI-шефа на Gemini.
- ТЗ-28: поле имени пользователя, поле названия блюда, кнопка «Искать» и область результата.
- Роли ADMIN, USER и GUEST ищутся по имени, незнакомое имя даёт GUEST.
- Ошибки показываются в модальном окне, а не в консоли.
- Сейчас цепочка такая: `index.html` → `main.js` → `authService.js`, `fileService.js`, `promptService.js`, `aiService.js`, `config.js`.

### Дерево компонентов и пропсы

::scheme fridge

Состояние живёт только в `App`: вниз уходят данные, а наверх приходят вызовы `onSearch`, `onDelete` и `onClose`.

| Компонент | Пропсы | Откуда приходят | Зачем нужен |
| --- | --- | --- | --- |
| `App` | `initialProducts`, `findRecipe` | Из `main.jsx` при запуске | Владелец состояния: пользователь, продукты, результат, ошибка |
| `Header` | `userName`, `role` | Из состояния `App` | Показывает, кто вошёл |
| `SearchForm` | `onSearch`, `isLoading` | От `App` | Поля из ТЗ-28; наверх отдаёт событие поиска |
| `ProductList` | `products`, `canDelete`, `onDelete` | От `App` | Список продуктов |
| `ProductCard` | `product`, `canDelete`, `onDelete` | От `ProductList`, транзитом из `App` | Одна карточка продукта |
| `ExpiryBadge` | `daysLeft` | От `ProductCard` | Плашка «Просрочено» или «Скоро истекает» |
| `RecipeResult` | `result`, `isLoading` | От `App` | Область результата из ТЗ-28 |
| `ErrorModal` | `isOpen`, `onClose`, `children` | От `App` | Ошибки в модальном окне |

### Полный код с комментариями

```jsx
// FridgeApp.jsx — набросок React-интерфейса «Умного холодильника» (ТЗ-28)
// Все данные приходят в компоненты через пропсы, состояние живёт в App.
import { useState } from 'react';

// Роли по ТЗ: роль ищется по имени, незнакомое имя — гость
export const ROLES = { ADMIN: 'ADMIN', USER: 'USER', GUEST: 'GUEST' };
const KNOWN_USERS = { alexey: ROLES.ADMIN, maria: ROLES.USER }; // пример данных

export function getRole(username) {
  return KNOWN_USERS[username.trim().toLowerCase()] ?? ROLES.GUEST;
}

// ---------- «Глупые» компоненты: только пропсы, без своего состояния ----------

function Header({ userName, role }) {
  return (
    <header>
      <h1>Умный холодильник</h1>
      {userName && <p className="who">Вы вошли как {userName} ({role})</p>}
    </header>
  );
}

function ExpiryBadge({ daysLeft }) {
  if (daysLeft < 0) return <span className="badge danger">Просрочено</span>;
  if (daysLeft <= 2) return <span className="badge warning">Скоро истекает</span>;
  return null; // срок в порядке — плашка не нужна
}

function ProductCard({ product, canDelete, onDelete }) {
  return (
    <li className="product-card">
      <span>{product.name} — {product.quantity} шт.</span>
      <ExpiryBadge daysLeft={product.daysLeft} />
      {canDelete && <button onClick={() => onDelete(product.id)}>Удалить</button>}
    </li>
  );
}

function ProductList({ products, canDelete, onDelete }) {
  if (products.length === 0) return <p>Холодильник пуст</p>;
  return (
    <ul className="product-list">
      {products.map((p) => (
        <ProductCard key={p.id} product={p} canDelete={canDelete} onDelete={onDelete} />
      ))}
    </ul>
  );
}

function RecipeResult({ result, isLoading }) {
  if (isLoading) return <p className="result">AI-шеф думает…</p>;
  if (!result) return null;
  return (
    <section className="result">
      <h2>{result.dish}</h2>
      <p>{result.text}</p>
    </section>
  );
}

function ErrorModal({ isOpen, onClose, children }) {
  if (!isOpen) return null;
  return (
    <div className="modal-backdrop" onClick={onClose}>
      <div className="modal" role="dialog" aria-modal="true" aria-label="Ошибка" onClick={(e) => e.stopPropagation()}>
        <h2>Ошибка</h2>
        <div className="modal-body">{children}</div>
        <button onClick={onClose}>Понятно</button>
      </div>
    </div>
  );
}

// ---------- Форма из ТЗ-28: поля хранит сама, наружу отдаёт только событие ----------

function SearchForm({ onSearch, isLoading }) {
  const [username, setUsername] = useState('');
  const [dish, setDish] = useState('');

  function handleSubmit(e) {
    e.preventDefault(); // не перезагружать страницу
    onSearch({ username: username.trim(), dish: dish.trim() });
  }

  return (
    <form onSubmit={handleSubmit}>
      <input name="username" value={username} onChange={(e) => setUsername(e.target.value)} placeholder="Имя пользователя" />
      <input name="dish" value={dish} onChange={(e) => setDish(e.target.value)} placeholder="Название блюда" />
      <button type="submit" disabled={isLoading}>{isLoading ? 'Ищем…' : 'Искать'}</button>
    </form>
  );
}

// ---------- Корень: единственный владелец состояния ----------

export default function App({ initialProducts, findRecipe }) {
  const [user, setUser] = useState(null); // { name, role }
  const [products, setProducts] = useState(initialProducts);
  const [result, setResult] = useState(null);
  const [error, setError] = useState(null);
  const [isLoading, setIsLoading] = useState(false);

  const canDelete = user?.role === ROLES.ADMIN; // производное значение: не храним, а вычисляем

  async function handleSearch({ username, dish }) {
    if (!username || !dish) {
      setError('Введите имя пользователя и название блюда.');
      return;
    }
    const role = getRole(username);
    setUser({ name: username, role });
    setIsLoading(true);
    try {
      const recipe = await findRecipe({ role, dish, products }); // сервис пришёл пропом
      setResult(recipe);
    } catch (err) {
      setError(err.message); // ошибку показываем в модальном окне, а не в консоли
    } finally {
      setIsLoading(false);
    }
  }

  function handleDelete(id) {
    setProducts((prev) => prev.filter((p) => p.id !== id));
  }

  return (
    <>
      <Header userName={user?.name} role={user?.role} />
      <SearchForm onSearch={handleSearch} isLoading={isLoading} />
      <ProductList products={products} canDelete={canDelete} onDelete={handleDelete} />
      <RecipeResult result={result} isLoading={isLoading} />
      <ErrorModal isOpen={error !== null} onClose={() => setError(null)}>
        <p>{error}</p>
      </ErrorModal>
    </>
  );
}
```

### Разбор решений

- **Состояние в одном месте.** `user`, `products`, `result`, `error` и `isLoading` нужны нескольким компонентам, поэтому живут в их общем родителе `App` (раздел 7).
- **Локальное состояние формы.** Набираемый текст нужен только `SearchForm`. Наружу форма отдаёт готовый результат через `onSearch({ username, dish })`.
- **Права как производное значение.** `canDelete` не хранится, а вычисляется из роли при каждом рендере. Запись `user?.role` — опциональная цепочка: если `user` равен `null`, результат будет `undefined` без ошибки.
- **Сервис приходит пропом.** `App` не знает, как устроен поиск рецепта: он просто вызывает `findRecipe`. Благодаря этому в проверке ниже вместо Gemini подставлена функция-заглушка.
- **Ошибки — в модальное окно.** И пустые поля, и ошибка сервиса попадают в `error`, а окно показывает текст через `children`.

### Запуск: откуда берутся initialProducts и findRecipe

```jsx
// main.jsx — точка входа вместо нынешнего main.js
import { createRoot } from 'react-dom/client';
import App from './FridgeApp.jsx';

// Запрос к своему серверу. Адрес /api/recipe условный — его ещё нужно сделать
async function findRecipe({ role, dish, products }) {
  const response = await fetch('/api/recipe', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ role, dish, products }),
  });
  if (!response.ok) throw new Error('AI-шеф не ответил. Попробуйте ещё раз.');
  return response.json(); // ожидаем { dish, text }
}

const initialProducts = [
  { id: 1, name: 'Молоко', quantity: 2, daysLeft: 1 },
  { id: 2, name: 'Яйца', quantity: 10, daysLeft: 7 },
  { id: 3, name: 'Йогурт', quantity: 1, daysLeft: -2 },
];

createRoot(document.getElementById('root')).render(
  <App initialProducts={initialProducts} findRecipe={findRecipe} />
);
```

Почему через сервер: в вашем `aiService.js` ключ Gemini читается из `.env` в Node.js. В браузере ключ увидел бы любой посетитель, поэтому запрос к AI должен идти с сервера. Структура продукта `{ id, name, quantity, daysLeft }` здесь условная: подстройте её под свой `fridge.json`.

### Что показала проверка

Код выше запущен в эмуляции браузера на React 19.3.1. Вместо настоящего AI подставлена заглушка. Она, как ваш `promptService.js`, бросает «Access denied» для всех ролей, кроме ADMIN и USER.

| Действие | Кто вошёл | Кнопок «Удалить» | Модальное окно | Результат |
| --- | --- | --- | --- | --- |
| Первый рендер | никто | 0 | нет | нет |
| «Искать» с пустыми полями | никто | 0 | «Введите имя пользователя и название блюда.» | нет |
| Имя «гость123», блюдо «омлет» | гость123 (GUEST) | 0 | «Access denied» | нет |
| Имя «Alexey», блюдо «омлет» | Alexey (ADMIN) | 3 | нет | «Омлет» с текстом рецепта |
| Админ удаляет «Яйца» | Alexey (ADMIN) | 2 | нет | в списке остались Молоко и Йогурт |

Плашки срока годности были видны сразу: «Скоро истекает» у молока и «Просрочено» у йогурта. Имя «Alexey» с большой буквы дало ADMIN, потому что `getRole` приводит имя к нижнему регистру.
